function Verify-PostDiffHashes($preHashes) {
    Write-Host "=== [4/6] Verificando hashes POST-DIFF (deben diferir de pre-diff donde corresponde) ==="
    $failures = 0
    foreach ($target in $AllTargets) {
        $f = Join-Path $TempRepo $target
        if (-not (Test-Path $f)) {
            # Para swarm/ dir, el objetivo es que NO exista
            if ($target -eq ".claude-flow/swarm/") {
                Write-Host "  $target: NO EXISTE (esperado — swarm/ retirado)"
                continue
            }
            throw "POST-DIFF: $target no existe pero debería"
        }
        $h = Get-SHA256 $f
        $orig = $preHashes[$target]
        if ($h -eq $orig) {
            Write-Error "HASH INVARIANTE: $target conserva hash original ($h). La transformación NO se aplicó."
            $failures++
        } else {
            Write-Host "  $target: CAMBIADO (pre=$orig, post=$h)"
        }
    }
    # Verificar swarm/ eliminado
    $swarmDir = Join-Path $TempRepo ".claude-flow\swarm"
    if (Test-Path $swarmDir) {
        Write-Error "swarm/ DIRECTORY AÚN EXISTE — debería haber sido retirado"
        $failures++
    } else {
        Write-Host "  .claude-flow/swarm/: CONFIRMADO ELIMINADO"
    }
    if ($failures -gt 0) { throw "$failures objetivo(s) fallaron verificación post-diff" }
    Write-Host "=== POST-DIFF: todos los hashes cambiaron como se esperaba ==="
}
